# workflows/weekly-short.yaml
name: Weekly short
schedule: "0 8 * * 1"   # mondays 8am
budget_usd: 5
agents: 4

stages:
  - swarm: research
    goal: >
      Find 5 short-form video ideas trending in the
      attraction niche this week. Check middle-out for
      what we already published and skip those. Each
      idea needs a hook line and a why.
    done_when: 5 ideas ranked by Erlich
    tools: [web_search, middle_out]

  - gate: pick
    ask: Pick one idea. Optional note for the script.

  - swarm: script
    goal: Write script and shot list for the pick, 20-25s.
    done_when: script.md and shots.md in deliverables
    tools: [middle_out]

  - connector: remotion_render
    input: deliverables/script.md
    output: deliverables/short.mp4

  - gate: review
    ask: Watch it. Approve or send back with a note.

  - connector: youtube_publish
    input: deliverables/short.mp4
    schedule_at: "next tuesday 09:00"
    visibility: scheduled
